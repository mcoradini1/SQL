Example 9: Deleting Tables (Drop table)
------------------------------------------------------------------------------------------------------------------------

create table if not exists test_table;                                -- command 1
id int,
name char(29),
age int,

insert into test_table;                                               -- command 2
('1', 'john' '22'),
('2', 'mary' '6'),
('3', 'dude' '81'),
('4', 'tape' '8');

select * from test_table;                                             -- command 3

drop table if exists test_table;                                      -- command 4
